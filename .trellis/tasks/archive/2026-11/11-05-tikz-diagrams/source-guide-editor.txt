import type { Node } from "prosemirror-model";
import type { NodeWithPos } from "../types";

/**
 * Finds the first image node in the document with the given src.
 *
 * @param doc - the document to search.
 * @param src - the image source URL to search for.
 * @returns the node and its position, or undefined.
 */
export function findImageNodeBySrc(
  doc: Node,
  src: string
): NodeWithPos | undefined {
  let found: NodeWithPos | undefined;
  doc.descendants((node, pos) => {
    if (found) {
      return false;
    }
    if (node.type.name === "image" && node.attrs.src === src) {
      found = { node, pos };
      return false;
    }
    return true;
  });
  return found;
}
```

然后把 `Diagrams.ts` 中的私有方法改为调用它（`findImageNodeBySrc(state.doc, srcToFind)`）。

### 扩展 —— `app/editor/extensions/Tikz.tsx`

```tsx
import { action, makeObservable, observable } from "mobx";
import type { Command } from "prosemirror-state";
import { NodeSelection } from "prosemirror-state";
import * as React from "react";
import type { CommandFactory, WidgetProps } from "@shared/editor/lib/Extension";
import Extension from "@shared/editor/lib/Extension";
import FileHelper, { ImageSource } from "@shared/editor/lib/FileHelper";
import { findImageNodeBySrc } from "@shared/editor/queries/findImageNodeBySrc";
import { DEFAULT_SOURCE } from "../lib/tikz/constants";
import { extractSource } from "../lib/tikz/svg";

const TikzEditor = React.lazy(() => import("../components/TikzEditor"));

type Session = {
  /** The src of the image being edited, or an empty string for a new one. */
  src: string;
  /** The TikZ source to show when the dialog opens. */
  source: string;
};

/**
 * Adds commands to create and edit TikZ diagrams. Diagrams are stored as SVG
 * images with the TikZ source embedded in their metadata.
 */
export default class TikzExtension extends Extension {
  get name() {
    return "tikz";
  }

  constructor(options = {}) {
    super(options);
    makeObservable(this);
  }

  commands(): Record<string, CommandFactory> {
    return {
      insertTikz: (): Command => (_state, dispatch) => {
        if (dispatch) {
          this.open({ src: "", source: DEFAULT_SOURCE });
        }
        return true;
      },
      editTikz: (): Command => (state, dispatch) => {
        const node =
          state.selection instanceof NodeSelection
            ? state.selection.node
            : undefined;
        if (
          node?.type.name !== "image" ||
          node.attrs.source !== ImageSource.Tikz
        ) {
          return false;
        }
        if (dispatch) {
          void this.openExisting(node.attrs.src);
        }
        return true;
      },
    };
  }

  widget = ({ readOnly }: WidgetProps) =>
    this.session && !readOnly ? (
      <React.Suspense fallback={null}>
        <TikzEditor
          initialSource={this.session.source}
          onSave={this.handleSave}
          onClose={this.close}
        />
      </React.Suspense>
    ) : undefined;

  /**
   * Reads the TikZ source out of an existing diagram and opens the dialog.
   *
   * @param src - the URL of the SVG to edit.
   */
  private async openExisting(src: string) {
    // See FileHelper.urlToBase64 for why the HTTP cache is skipped.
    const response = await fetch(src, { cache: "no-store" });
    const source = response.ok
      ? extractSource(await response.text())
      : undefined;
    this.open({ src, source: source ?? DEFAULT_SOURCE });
  }

  /**
   * Uploads the rendered SVG and inserts it, or replaces the image that was
   * being edited.
   *
   * @param svg - the SVG markup with the TikZ source embedded.
   */
  private handleSave = async (svg: string) => {
    const session = this.session;
    const { uploadFile } = this.editor.props;
    if (!session || !uploadFile) {
      throw new Error("No upload handler configured");
    }

    const file = new File([svg], "diagram.svg", { type: "image/svg+xml" });
    const [src, dimensions] = await Promise.all([
      uploadFile(file),
      FileHelper.getImageDimensions(file),
    ]);

    // Read fresh state, the document may have changed during the upload.
    const { state, dispatch } = this.editor.view;
    const attrs = { ...(dimensions ?? {}), src, source: ImageSource.Tikz };
    const existing = session.src
      ? findImageNodeBySrc(state.doc, session.src)
      : undefined;

    if (existing) {
      dispatch(
        state.tr.setNodeMarkup(existing.pos, undefined, {
          ...existing.node.attrs,
          ...attrs,
        })
      );
    } else {
      dispatch(
        state.tr.replaceSelectionWith(
          this.editor.schema.nodes.image.create(attrs)
        )
      );
    }
    this.close();
  };

  @action
  private open(session: Session) {
    this.session = session;
  }

  @action
  private close = () => {
    this.session = undefined;
    this.editor.view.focus();
  };

  @observable
  private session: Session | undefined;
}
```

### 设计说明

- **为什么用 widget。** 与 `FindAndReplaceExtension` 一样：widget 会被自动 `observer` 包装，`session` 变化即重渲染。
- **读者不加载任何东西。** `TikzEditor` 是 `React.lazy`，只有打开对话框时才下载它和 `threads`。
- **每次保存都上传新附件。** 与 diagrams.net 相同；旧附件仍由文档历史引用，不要删除。

## 第 7 步 —— TikZ 编辑对话框

对话框左侧是代码，右侧是预览，下方是 TeX 错误信息。代码编辑器与插件 `src/editor.ts` 相同，用 CodeMirror 6 和 `codemirror-lang-latex`（第 0 步已安装）。因为整个文件通过 `React.lazy` 加载，CodeMirror 不会进入主包。

新建 `app/editor/components/TikzEditor.tsx`：

```tsx
import { defaultKeymap, history, historyKeymap } from "@codemirror/commands";
import { EditorState } from "@codemirror/state";
import { EditorView, keymap, lineNumbers } from "@codemirror/view";
import { latex } from "codemirror-lang-latex";
import * as React from "react";
import { useTranslation } from "react-i18next";
import styled from "styled-components";
import { s } from "@shared/styles";
import Button from "~/components/Button";
import Flex from "~/components/Flex";
import Modal from "~/components/Modal";
import { renderTikz } from "../lib/tikz/renderTikz";

/** Delay between the last keystroke and the preview render. */
const PREVIEW_DELAY_MS = 600;

type Props = {
  /** The TikZ source to start with. */
  initialSource: string;
  /** Called with the final SVG when the user saves. */
  onSave: (svg: string) => Promise<void>;
  /** Called when the dialog is dismissed. */
  onClose: () => void;
};

/**
 * A dialog with a TikZ code editor and a live preview rendered in a worker.
 */
export default function TikzEditor({ initialSource, onSave, onClose }: Props) {
  const { t } = useTranslation();
  const editorRef = React.useRef<HTMLDivElement>(null);
  const [source, setSource] = React.useState(initialSource);
  const [svg, setSvg] = React.useState<string>();
  const [error, setError] = React.useState<string>();
  const [rendering, setRendering] = React.useState(false);
  const [saving, setSaving] = React.useState(false);

  // Mount CodeMirror once; it owns the text from then on.
  React.useEffect(() => {
    if (!editorRef.current) {
      return;
    }
    const view = new EditorView({
      parent: editorRef.current,
      state: EditorState.create({
        doc: initialSource,
        extensions: [
          lineNumbers(),
          history(),
          keymap.of([...defaultKeymap, ...historyKeymap]),
          latex(),
          EditorView.updateListener.of((update) => {
            if (update.docChanged) {
              setSource(update.state.doc.toString());
            }
          }),
        ],
      }),
    });
    view.focus();
    return () => view.destroy();
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, []);

  // Re-render after typing stops, aborting any render still in flight.
  React.useEffect(() => {
    const controller = new AbortController();
    const timer = setTimeout(async () => {
      setRendering(true);
      const result = await renderTikz(source, controller.signal);
      if (controller.signal.aborted) {
        return;
      }
      setRendering(false);
      if (result.ok) {
        setSvg(result.svg);
        setError(undefined);
      } else {
        setError(result.error);
      }
    }, PREVIEW_DELAY_MS);

    return () => {
      clearTimeout(timer);
      controller.abort();
    };
  }, [source]);

  const handleSave = async () => {
    if (!svg) {
      return;
    }
    setSaving(true);
    try {
      await onSave(svg);
    } catch (err) {
      setError(err instanceof Error ? err.message : String(err));
    } finally {
      setSaving(false);
    }
  };

  return (
    <Modal
      isOpen
      title={t("TikZ diagram")}
      width="min(1100px, 90vw)"
      onRequestClose={onClose}
    >
      <Flex gap={16}>
        <Code ref={editorRef} />
        <Preview align="center" justify="center">
          {svg ? (
            <img
              src={`data:image/svg+xml;charset=utf-8,${encodeURIComponent(svg)}`}
              alt={t("Preview")}
            />
          ) : null}
          {rendering && <Status>{t("Rendering…")}</Status>}
        </Preview>
      </Flex>
      {error && <ErrorMessage role="alert">{error}</ErrorMessage>}
      <Flex justify="flex-end" gap={8} style={{ marginTop: 16 }}>
        <Button onClick={onClose} neutral>
          {t("Cancel")}
        </Button>
        <Button
          onClick={handleSave}
          disabled={!svg || !!error || rendering || saving}
        >
          {saving ? `${t("Saving")}…` : t("Save")}
        </Button>
      </Flex>
    </Modal>
  );
}

const Code = styled.div`
  flex: 1;
  min-width: 0;
  height: 60vh;
  overflow: auto;
  border: 1px solid ${s("inputBorder")};
  border-radius: 4px;
  font-size: 14px;
`;

const Preview = styled(Flex)`
  flex: 1;
  position: relative;
  height: 60vh;
  overflow: auto;
  border: 1px solid ${s("inputBorder")};
  border-radius: 4px;
  background: white;

  img {
    max-width: 100%;
  }
`;

const Status = styled.div`
  position: absolute;
  top: 8px;
  right: 8px;
  font-size: 13px;
  color: ${s("textTertiary")};
`;

const ErrorMessage = styled.pre`
  margin: 12px 0 0;
  max-height: 120px;
  overflow: auto;
  white-space: pre-wrap;
  font-size: 13px;
  color: ${s("danger")};
`;
```

### 设计说明

- **预览用 `data:` URL。** 与最终显示方式一致（`<img>`），字体是否嵌入正确在预览里就能看出来。
- **预览背景固定为白色。** TikZ 默认用黑色绘图，暗色主题下的显示在第 9 步单独处理。
- **保存按钮在渲染中禁用。** 否则保存的 SVG 可能对应旧的代码。
- **每次输入都中止上一次渲染。** `controller.abort()` 会触发 `renderTikz` 中的 `Thread.terminate`，同一时间最多只有一个 worker 在运行。

## 第 9 步 —— 暗色模式与翻译

### 暗色模式

TikZ 默认用黑色绘图、背景透明，在暗色主题下几乎看不见。`<img>` 中的 SVG 无法读取页面的 CSS 变量，所以在外面用滤镜：`invert(1)` 把黑变白，`hue-rotate(180deg)` 把彩色大致转回原来的色相。

先让图片带上来源。在 `shared/editor/components/Image.tsx` 中，给显示图片的 `<img className={EditorStyleHelper.imageHandle}>`（带链接和不带链接两个分支）加一个属性：

```tsx
              data-source={node.attrs.source}
```

然后在 `shared/editor/components/Styles.ts` 中、`.image .image-wrapper img` 规则附近加：

```typescript
.image img[data-source="tikz"] {
  filter: ${(props) =>
    props.theme.isDark ? "invert(1) hue-rotate(180deg)" : "none"};
}
```

读者视图和编辑器用的是同一个组件和样式，一处修改即可。HTML/PDF 导出始终是浅色，不受影响。
